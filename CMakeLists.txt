cmake_minimum_required(VERSION 3.16)
project(NetRescue VERSION 1.0.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

add_library(netrescue_core OBJECT src/network.c src/cidr.c src/persistence.c src/arq.c src/topology.c src/congestion.c)
target_include_directories(netrescue_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(UNIX)
  target_link_libraries(netrescue_core PUBLIC m)
endif()
if(MSVC)
  target_compile_options(netrescue_core PRIVATE /W4)
else()
  target_compile_options(netrescue_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
  add_executable(NetRescue WIN32 src/main_win32.c)
  target_link_libraries(NetRescue PRIVATE netrescue_core comctl32 comdlg32 user32 gdi32)
  if(MSVC)
    target_compile_options(NetRescue PRIVATE /W4)
  else()
    target_compile_options(NetRescue PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_executable(NetRescueCLI src/main_cli.c)
  target_link_libraries(NetRescueCLI PRIVATE netrescue_core)
  if(MSVC)
    target_compile_options(NetRescueCLI PRIVATE /W4)
  else()
    target_compile_options(NetRescueCLI PRIVATE -Wall -Wextra -Wpedantic)
  endif()
else()
  message(STATUS "Native Win32 dashboard is Windows-only; building the portable simulator CLI.")
  add_executable(NetRescue src/main_cli.c)
  target_link_libraries(NetRescue PRIVATE netrescue_core)
  target_compile_options(NetRescue PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
add_executable(netrescue_tests tests/test_core.c)
target_link_libraries(netrescue_tests PRIVATE netrescue_core)
if(MSVC)
  target_compile_options(netrescue_tests PRIVATE /W4 /UNDEBUG)
else()
  target_compile_options(netrescue_tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
endif()
add_test(NAME core COMMAND netrescue_tests)
